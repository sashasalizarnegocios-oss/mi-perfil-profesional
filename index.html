function mostrarHabilidades() { let habilidades = document.getElementById("contenido-habilidades"); let boton = document.getElementById("boton-habilidades"); if (habilidades.style.display === "none") { habilidades.style.display = "block"; boton.textContent = "Ocultar habilidades"; boton.setAttribute("aria-expanded", "true"); } else { habilidades.style.display = "none"; boton.textContent = "Mostrar habilidades"; boton.setAttribute("aria-expanded", "false"); } }
